// [02] FontWeight4  (int 100-900)
private fun FontWeight4() {
    val items = listOf("100","200","300","400","500","600","700","800","900")
    val current = (sp.getInt("FontWeight4", 400) / 100) - 1
    AppPopup.singleChoice(
        context = this,
        title = "FontWeight4",
        items = items,
        checkedIndex = current,
    ) { i, _ ->
        sp.edit().putInt("FontWeight4", (i + 1) * 100).apply()
    }
}
